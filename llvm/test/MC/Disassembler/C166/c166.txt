# RUN: llvm-mc -triple=c166 -disassemble %s | FileCheck %s

# The opcode byte alone decides whether an instruction is one or two words.

# CHECK: add r2, r3
0x00 0x23
# CHECK: addb rl2, rh3
0x01 0x47
# CHECK: add r2, #1234
0x06 0xf2 0xd2 0x04
# CHECK: sub r4, r5
0x20 0x45
# CHECK: cmp r2, r3
0x40 0x23
# CHECK: cpl r3
0x91 0x30
# CHECK: neg r3
0x81 0x30

# The shift count of an immediate shift is in the high nibble.
# CHECK: shl r2, #3
0x5c 0x32
# CHECK: ashr r2, #2
0xbc 0x22

# MOVBZ and MOVBS reverse the usual nibble order.
# CHECK: movbz r4, rl5
0xc0 0xa4
# CHECK: movbs r6, rh7
0xd0 0xf6

# DIV repeats the register number in both nibbles.
# CHECK: mul r2, r3
0x0b 0x23
# CHECK: div r3
0x4b 0x33
# CHECK: divu r5
0x5b 0x55

# CHECK: mov r2, r3
0xf0 0x23
# CHECK: mov r2, #4660
0xe6 0xf2 0x34 0x12
# CHECK: mov r2, [r3+#8]
0xd4 0x23 0x08 0x00
# CHECK: mov [r0+#4], r2
0xc4 0x20 0x04 0x00
# CHECK: mov r2, [r3]
0xd4 0x23 0x00 0x00

# A conditional jump keeps its condition code.
# CHECK: jmpa cc_ULT, 4660
0xea 0x80 0x34 0x12
# CHECK: jmpa cc_UC, 4660

# CHECK: calla cc_UC, 4660
0xca 0x00 0x34 0x12
# CHECK: calli cc_UC, [r5]
0xab 0x05
# CHECK: jmpi cc_UC, [r6]
0x9c 0x06

# CHECK: reti
