# RUN: llvm-mc -triple=c166 -disassemble %s | FileCheck %s

# The opcode byte alone decides whether an instruction is one or two words.

# CHECK: add r2, r3
0x00 0x23
# CHECK: addb rl2, rh3
0x01 0x47
# CHECK: add r2, #1234
0x06 0xf2 0xd2 0x04
# CHECK: sub r4, r5
0x20 0x45
# CHECK: cmp r2, r3
0x40 0x23
# CHECK: cpl r3
0x91 0x30
# CHECK: neg r3
0x81 0x30

# The shift count of an immediate shift is in the high nibble.
# CHECK: shl r2, #3
0x5c 0x32
# CHECK: ashr r2, #2
0xbc 0x22

# MOVBZ and MOVBS reverse the usual nibble order.
# CHECK: movbz r4, rl5
0xc0 0xa4
# CHECK: movbs r6, rh7
0xd0 0xf6

# DIV repeats the register number in both nibbles.
# CHECK: mul r2, r3
0x0b 0x23
# CHECK: div r3
0x4b 0x33
# CHECK: divu r5
0x5b 0x55

# CHECK: mov r2, r3

# CHECK: mov r2, #4660

# A bit-addressable special function register comes back by name, because its
# bitoff is the short address the register file already carries.
# CHECK: bclr mdc.0
0x0e 0x87
# CHECK: bset s0con.15
0xff 0xd8
# Bit-addressable RAM has nothing mapped to name, so it stays a number.
# CHECK: bset 0.3
0x3f 0x00
# CHECK: band r2.1, r3.2
0x6a 0xf3 0xf2 0x21
# CHECK: bfldl r4, #15, #5
0x0a 0xf4 0x0f 0x05

# The bit test branches print the distance in words, which is what can be fed
# back to the assembler; llvm-objdump asks for the target address instead.
# CHECK: jb r5.3, -3
0x8a 0xf5 0xfd 0x30
# CHECK: jnb psw.10, 4

# CHECK: jbc r5.3, 2

# The two byte relative jump.  The condition is the high nibble of the opcode
# and the displacement counts words from the instruction after it, so -1
# branches to the instruction itself; that is what LLD writes as padding.
# CHECK: jmpr cc_UC, -1
0x0d 0xff
# CHECK: jmpr cc_EQ, 4
0x2d 0x04
# CHECK: jmpr cc_ULE, -128
0xfd 0x80

# An address where one of the special function registers is mapped is named
# rather than spelled out, and the assembler takes the name back as the same
# address, so this still reassembles to the bytes it came from.
# CHECK: mov r2, mdl
0xf2 0xf2 0x0e 0xfe
# CHECK: mov mdh, r3
0xf6 0xf3 0x0c 0xfe
# CHECK: mov r1, stkov
0xf2 0xf1 0x14 0xfe
# CHECK: mov psw, r5
0xf6 0xf5 0x10 0xff
# Anything else stays a number.
# CHECK: mov r4, 4660
0xf2 0xf4 0x34 0x12

# The 8 bit "reg" field of an arithmetic instruction reaches the special
# function registers, and comes back by name.
# CHECK: add mdl, #1
0x06 0x07 0x01 0x00
# CHECK: add r2, #1234
0x06 0xf2 0xd2 0x04
# CHECK: sub stkov, #256
0x26 0x0a 0x00 0x01
# CHECK: mov sp, #64512
0xe6 0x09 0x00 0xfc
# CHECK: mov mdl, mdh
0xf2 0x07 0x0c 0xfe

# TRAP carries half its vector's address, so these come back as the trap
# numbers they were written as.
# CHECK: trap #2
0x9b 0x04
# CHECK: trap #127
0x9b 0xfe

# In a byte instruction the same field names a byte register instead, so these
# opcodes come back with the byte mnemonic and the byte register name.
# CHECK: addb mdl, #1

# CHECK: addb rl2, #200
0x07 0xf4 0xc8 0x00
# CHECK: subb stkov, #16

# CHECK: cmpb mdl, #0

# CHECK: movb dpp0, #5
0xe7 0x00 0x05 0x00
# CHECK: movb mdl, mdh
0xf3 0x07 0x0c 0xfe
